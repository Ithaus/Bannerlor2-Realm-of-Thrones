    class KsPart
    {
        public byte[] C; public float[] X; public float Rent; public KsBlock K; public double Hands;
        public double[] LC; public int Seed; public static int NanEvery;
        public double L(int c, string type)
        {
            if (LC == null) { LC = new double[KsTab.NC]; for (int i = 0; i < LC.Length; i++) LC[i] = double.NaN; }
            double v = LC[c];
            if (double.IsNaN(v))
            {
                uint h = (uint)(Seed * 2654435761u) ^ (uint)(c * 40503u) ^ (uint)(type != null ? type.Length * 977 : 0);
                h ^= h >> 13; h *= 0x5bd1e995; h ^= h >> 15;
                v = (h % 10000) / 3000.0 - ((h % 37) == 0 ? 0.5 : 0.0);
                if (NanEvery > 0 && (h % (uint)NanEvery) == 0) v = double.NaN;   // proba NaN (LandEff zepsuty) - ten sam NaN co dotad
                LC[c] = v;
            }
            return v;
        }
    }
